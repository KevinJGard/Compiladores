    \item De las gramáticas que se muestran a continuación, reescríbelas siguiendo las indicaciones de asociación y precedencia que se indican para los operadores:
    \begin{enumerate}[label=(\alph*)]
        \item La operación :: asocia por la derecha; + y - asocian por la izquierda. El operador :: precede a + y -; + y - tienen la misma precedencia.
        \begin{verbatim}
            L := L :: L | L + element | L - element |
        \end{verbatim}
        donde element es un terminal (dado por el lexer) y la última regla refiere al elemento vacío.
        \item $>$ asocia por la derecha; ; asocia por la izquierda; ° asocia por la derecha; $\#$ no asociativo. ; precede a $>$ que precede a $\#$ que precede a °
        \begin{verbatim}
            S := S > S | S ; S | S > S | S # S | a | b
        \end{verbatim}
        Los terminales son $a$ y $b$ más los operadores indicados:
    \end{enumerate}

\begin{solucion}


\medskip\noindent\textbf{\underline{Inciso (a)}}

\[
L \to L \occ L \mid L + \tk{element} \mid L - \tk{element} \mid \varepsilon
\]

\textbf{Niveles.} $+$ y $-$ tienen la misma precedencia y asocian por la
izquierda; $\occ$ liga más fuerte y asocia por la derecha. Por tanto $+$ y $-$
van arriba (mismo nivel) y $\occ$ abajo.

\textbf{Gramática reescrita:}
\[
\begin{array}{rcl}
L &\to& L + M \mid L - M \mid M \mid \varepsilon \\
M &\to& N \occ M \mid N \\
N &\to& \tk{element}
\end{array}
\]
 ejemplo

\begin{itemize}[leftmargin=*]
  \item $L \to L + M$: recursión a la izquierda, así que $+$ y $-$ asocian por
        la izquierda: $\tk{e} - \tk{e} - \tk{e}$ se agrupa como
        $(\tk{e} - \tk{e}) - \tk{e}$.
 
\end{itemize}

\textbf{Ambiguedad.} En la gramática original la cadena
$\occ\ \occ$ tiene 2 árboles, y la cadena
$+\,\tk{element} \occ -\,\tk{element} \occ +\,\tk{element}$ tiene 7. En la
reescrita, toda cadena tiene un único árbol.



\medskip\noindent\textbf{\underline{Inciso (b)}}

 La gramática es:
\[
S \to S \ogt S \mid S \osc S \mid S \odg S \mid S \ohs S \mid \tk{a} \mid \tk{b}
\]
donde $\odg$ denota el operador ``$^\circ$''.

\textbf{Niveles.} La cadena de precedencia es
``$\osc$ precede a $\ogt$, que precede a $\ohs$, que precede a $\odg$''. De
más flojo (arriba) a más fuerte (abajo):

\begin{center}
\begin{tabular}{c c l l}
\toprule
Nivel & Operador & Asociatividad & Forma \\
\midrule
$S$ & $\odg$ & derecha       & $S \to T \odg S \mid T$ \\
$T$ & $\ohs$ & no asociativo & $T \to U \ohs U \mid U$ \\
$U$ & $\ogt$ & derecha       & $U \to V \ogt U \mid V$ \\
$V$ & $\osc$ & izquierda     & $V \to V \osc W \mid W$ \\
$W$ & ---    & ---           & $W \to \tk{a} \mid \tk{b}$ \\
\bottomrule
\end{tabular}
\end{center}

\textbf{Gramática reescrita:}
\[
\begin{array}{rcl}
S &\to& T \odg S \mid T \\
T &\to& U \ohs U \mid U \\
U &\to& V \ogt U \mid V \\
V &\to& V \osc W \mid W \\
W &\to& \tk{a} \mid \tk{b}
\end{array}
\]

\textbf{Ejemplos de agrupación} :

\begin{tabular}{l l c}
\toprule
Cadena & Agrupación & Árboles en la original \\
\midrule

$\tk{a} \osc \tk{a} \osc \tk{a}$ & $(\tk{a} \osc \tk{a}) \osc \tk{a}$ & 2 \\
$\tk{a} \osc \tk{b} \ogt \tk{a}$ & $(\tk{a} \osc \tk{b}) \ogt \tk{a}$ & 2 \\

$\tk{a} \ohs \tk{b} \odg \tk{a} \ohs \tk{b}$ & $(\tk{a} \ohs \tk{b}) \odg (\tk{a} \ohs \tk{b})$ & 5 \\

\bottomrule

\medskip\noindent\textbf{\underline{Inciso (a): forma normal de Chomsky}}

Primero sacamos los terminales de las reglas largas. En una regla de
longitud $\geq 2$ no pueden aparecer terminales. Se crea un no terminal para
cada uno:
\[
O_{+} \to +, \quad O_{-} \to -, \quad O_{*} \to *, \quad O_{/} \to /, \quad
A \to (, \quad C \to )
\]
con lo que
\[
E \to E\,O_{+}\,E \mid E\,O_{-}\,E \mid E\,O_{*}\,E \mid E\,O_{/}\,E \mid A\,E\,C \mid \tk{num}
\]

Después se aplica la regla de los tres símbolos: cada regla
$E \to E\,O_{+}\,E$ se divide en $E \to E\,R_{+}$ y $R_{+} \to O_{+}\,E$, donde
$R_{+}$ representa ``el resto de una suma'': el operador seguido del segundo
operando. Igual con los demás operadores. Para el paréntesis,
$E \to A\,E\,C$ se divide en $E \to A\,X$ y $X \to E\,C$, donde $X$ es ``lo que
sigue a un paréntesis que abre''.

\textbf{Gramática en FNC} (12 no terminales, 17 producciones):
\[
\begin{array}{rcl}
E &\to& E\,R_{+} \mid E\,R_{-} \mid E\,R_{*} \mid E\,R_{/} \mid A\,X \mid \tk{num} \\[2pt]
R_{+} &\to& O_{+}\,E \qquad R_{-} \to O_{-}\,E \qquad R_{*} \to O_{*}\,E \qquad R_{/} \to O_{/}\,E \\[2pt]
X &\to& E\,C \\[2pt]
O_{+} &\to& + \qquad O_{-} \to - \qquad O_{*} \to * \qquad O_{/} \to / \qquad A \to ( \qquad C \to )
\end{array}
\]

    \item Dada la producción $X \rightarrow \alpha Y \beta$, demostrar que si \textsc{Null}$(\beta)$, entonces \textsc{Follow}$(X)$ $\subseteq$ \textsc{Follow}$(Y)$.

\begin{solucion}
\begin{lema}
Si $\beta \deriva \varepsilon$, entonces para cualesquiera cadenas $\gamma$ y
$\delta$ se cumple $\gamma\,\beta\,\delta \deriva \gamma\,\delta$.
\end{lema}

\begin{proof}
En una gramática libre de contexto cada no terminal se reescribe sin importar
lo que tenga alrededor. Los mismos pasos que llevan $\beta$ a $\varepsilon$
pueden aplicarse dentro de $\gamma\,\beta\,\delta$ sin tocar $\gamma$ ni
$\delta$.
\end{proof}

\begin{prop}
Sea $X \to \alpha\, Y\, \beta$ una producción. Si $\Null(\beta)$, entonces
$\Follow(X) \subseteq \Follow(Y)$.
\end{prop}

\begin{proof}
Sea $t \in \Follow(X)$. Por definición existen $\gamma, \delta$ tales que
\[
S\,\$ \deriva \gamma\, X\, t\, \delta. \tag{1}
\]
Aplicando la producción $X \to \alpha\, Y\, \beta$:
\[
\gamma\, X\, t\, \delta \Rightarrow \gamma\, \alpha\, Y\, \beta\, t\, \delta. \tag{2}
\]
Como $\Null(\beta)$, por el lema (con contexto izquierdo $\gamma\alpha Y$ y
contexto derecho $t\delta$):
\[
\gamma\, \alpha\, Y\, \beta\, t\, \delta \deriva \gamma\, \alpha\, Y\, t\, \delta. \tag{3}
\]
Encadenando (1), (2) y (3):
\[
S\,\$ \deriva (\gamma\alpha)\, Y\, t\, \delta.
\]
Así, $t$ aparece inmediatamente a la derecha de $Y$ en una forma sentencial,
y por definición $t \in \Follow(Y)$. Como $t$ era un elemento arbitrario de
$\Follow(X)$, se concluye $\Follow(X) \subseteq \Follow(Y)$.
\end{proof}
\end{solucion}
